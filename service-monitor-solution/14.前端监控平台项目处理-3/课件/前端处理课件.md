```typescript
import axios, { CreateAxiosDefaults } from "axios";

const config: CreateAxiosDefaults = {
  baseURL: "/api",
  timeout: 5000
};

export const request = axios.create(config);

// 自动将本地存储的 token 添加到请求头Authorization中
request.interceptors.request.use(config => {
  const token = localStorage.getItem("token");
  if (token) {
    config.headers.Authorization = "Bearer " + token;
  }
  return config;
});

// 如果返回权限不足，跳转到登录页
request.interceptors.response.use(
  response => {
    return response.data;
  },
  error => {
    if (error.response.status === 401) {
    	window.location.href = "/login";
    }
    return Promise.reject(error);
  }
);

```

### 权限判定

有了登录之后，当然就可以有基本的权限控制，如果没有登录，是不能访问到比如根目录路由的，当然子路由肯定也访问不到了，所以针对根路由，我们可以做一个简单的权限判断。

在router目录下创建权限组件，`AuthRoute.tsx`

```typescript
import React from "react";
import { Navigate } from "react-router-dom";

interface AuthRouteProps {
  children: React.ReactNode;
}

const AuthRoute = ({ children }: AuthRouteProps) => {
  // 如果用户没有登录，重定向到登录页面
  if (!localStorage.getItem("token")) {
    return <Navigate to={`login?redirect=${window.location.pathname}`} />;
  }

  // 如果用户已登录，渲染子组件
  return children;
};

export default AuthRoute;

```

#### TanStack Query

然后在导航上我们需要读取用户相关信息，不过在React中使用`useEffect`钩子，来获取数据时，开发者必须手动处理诸如**数据加载状态**、**数据加载失败时的错误状态**、**组件卸载时取消请求**、**更新组件状态**、**缓存**等操作

管理这些不同的任务和边界情况可能会很复杂且耗时，这里使用[TanStack Query](https://tanstack.com/query/latest/docs/framework/react/overview)库（又叫做React Query），这个库提供了更简单的API，使得获取数据、管理加载和错误状态以及更新组件状态变得容易

使用useEffect通常我们会有下面的一些痛点：

- 重复请求地狱：用户反复切页面，每次都重新调接口
- 加载状态混乱：页面一会显示加载中，一会显示暂无数据
- 错误处理麻烦：接口报错得自己写try/catch，还得考虑重试，代码里全是if (error) { ... }
- 缓存处理冗余：刚看过的商品列表，切到详情页再回来又得重新加载

**TanStack Query相当于就是把上面的把数据管理从”手动体力活“变成”自动智能服务“**

1. **自动缓存**

`useEffect`调接口后，数据得自己存到`useState`里，下次用还得判断这数据是不是刚取过。TanStack Query自带**全局缓存**，同样的请求只会发一次，后续直接从缓存取

2. **请求去重**

用户快速切页面时，`useEffect`会触发多次请求，导致服务器压力大。TanStack Query的**请求去重**功能会合并相同请求，只发一次

3. **生命周期感知**

用`useEffect`时，页面切走再切回来，可能触发不必要的请求（比如用户切到其他标签页，回来时数据又重新加载）。TanStack Query能感知组件**挂载/卸载状态**，页面不可见时自动暂停请求，恢复时再继续

4. **加载状态处理与错误重试**

`useEffect`里的错误处理得自己写`try/catch`，加载状态得用`isLoading`变量控制，代码里全是`if`判断。TanStack Query自带**自动错误重试**（比如接口500，自动重试3次）和**统一的状态管理**（`isLoading`/`isError`/`isSuccess`)

**伪代码对比示例：**

**使用`useEffect`**

```typescript
import React, { useState, useEffect } from 'react';
import axios from 'axios';

function ProductList() {
  // 数据状态：加载中、数据、错误
  const [products, setProducts] = useState([]);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState(null);

  // 用useEffect触发请求
  useEffect(() => {
    const fetchProducts = async () => {
      setIsLoading(true); // 开始加载
      setError(null); // 清空错误
      try {
        const response = await axios.get('/api/products');
        setProducts(response.data); // 保存数据
      } catch (err) {
        setError(err.message); // 保存错误信息
        // 手动重试？得自己写定时器，麻烦！
      } finally {
        setIsLoading(false); // 加载结束
      }
    };

    fetchProducts();
  }, []); 

  // 手动处理各种状态
  if (isLoading) return <div>加载中...</div>;
  if (error) return <div>出错啦：{error}</div>;
  return (
    <ul>
      {products.map(product => (
        <li key={product.id}>{product.name}</li>
      ))}
    </ul>
  );
}
```

而且上面的代码，没有缓存，页面切走再切回，数据会重新加载，接口报错得自己处理重试

使用TanStack Query只需要三步：

1. **安装：**

```typescript
pnpm add @tanstack/react-query
```

2. **配置全局管家**

```typescript
import React from 'react';
import { createRoot } from 'react-dom/client';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import ProductList from './ProductList';

// 创建Query Client（数据管家）
const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      retry: 3, // 自动重试3次
      staleTime: 5 * 60 * 1000, // 缓存5分钟内不过期
    },
  },
});

const root = createRoot(document.getElementById('root'));
root.render(
  <QueryClientProvider client={queryClient}>
    <ProductList />
  </QueryClientProvider>
);
```

3. **组件中使用useQuery**

```typescript
import React from 'react';
import { useQuery } from '@tanstack/react-query';
import axios from 'axios';

function ProductList() {
  // useQuery自动管理请求（参数1：缓存键，参数2：请求函数）
  const { 
    data: products, // 数据（加载成功时）
    isLoading, // 加载中状态
    isError, // 错误状态
    error, // 错误信息
  } = useQuery({
    queryKey: ['products'], // 缓存键（唯一标识这个请求）
    queryFn: () => axios.get('/api/products').then(res => res.data), // 请求函数
  });

  // React Query自动提供状态，不用自己写if
  if (isLoading) return <div>加载中...</div>;
  if (isError) return <div>出错啦：{error.message}</div>;
  return (
    <ul>
      {products.map(product => (
        <li key={product.id}>{product.name}</li>
      ))}
    </ul>
  );
}
```

- **自动缓存**：相同的queryKey（['products']）再次请求时，直接从缓存取数据，不用重新调接口；
- **自动重试**：接口报错自动重试3次（全局配置中设置），不用自己写try/catch和重试逻辑；
- **生命周期感知**：页面切走时暂停请求，切回来时如果缓存没过期，直接显示缓存数据；
- **加载状态统一**：isLoading/isError是Tanstack Query自动维护的，不用自己用useState管理。

在我们的测试项目中，之前都是直接上报，现在既然有不同的项目，就应该根据不同的项目id进行上报，比如我们在demos的测试项目中稍微修改一下：

**main.ts**

```typescript
init({
  dsn: "/dsn-api/storage/tracing/vanillaLyLnF_"  // vanillaLyLnF_是具体的项目appId
});
```

当然，加上这个路径Params之后，`dsn-server`项目中的代码肯定也需要进行修改：

**storage.controller.ts**

```typescript
@Post("tracing/:app_id")
async tracing(@Param() { app_id }: { app_id: string }, @Body() body: any) {
  await this.storageService.tracing(app_id, body);
  return { message: "Tracing data inserted successfully" };
}
```

**storage.service.ts**

```typescript
async tracing(app_id: string, body: any) {
  const values = {
    app_id: app_id || "1",
    event_type: body.event_type || "default_event",
    message: body.message || "No message provided",
    info: body || {}
  };

  await this.clickhouseClient.insert({
    table: "monitor_storage",
    columns: ["app_id", "event_type", "message", "info"],
    values,
    format: "JSONEachRow"
  });
}
```
